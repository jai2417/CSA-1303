#include <stdio.h>
#include <string.h>

int main()
{
    char str[100];
    int i, valid = 1;

    printf("Enter a string: ");
    scanf("%99s", str);

    if (str[0] != 'a')
        valid = 0;

    for (i = 0; str[i] != '\0'; i++)
    {
        if (str[i] != 'a' && str[i] != 'b')
            valid = 0;
    }

    if (str[strlen(str) - 1] != 'a')
        valid = 0;

    if (valid)
        printf("String Accepted");
    else
        printf("String Rejected");

    return 0;
}
