#include <stdio.h>
#include <string.h>

int main()
{
    char str[100];
    int i, valid = 1;

    printf("Enter a binary string: ");
    scanf("%99s", str);

    if (strlen(str) < 2)
        valid = 0;

    if (str[0] != '0' || str[strlen(str) - 1] != '1')
        valid = 0;

    for (i = 0; str[i] != '\0'; i++)
    {
        if (str[i] != '0' && str[i] != '1')
            valid = 0;
    }

    if (valid)
        printf("String belongs to the CFG");
    else
        printf("String does not belong to the CFG");

    return 0;
}
OUTPUT:
Enter a binary string: 0101
String belongs to the CFG
