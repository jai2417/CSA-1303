#include <stdio.h>
#include <string.h>

int main()
{
    char str[100];
    int i, valid = 1;

    printf("Enter a binary string: ");
    scanf("%99s", str);

    if (str[0] != '0')
        valid = 0;

    for (i = 0; str[i] != '\0'; i++)
    {
        if (str[i] != '0' && str[i] != '1')
        {
            valid = 0;
            break;
        }
    }

    if (str[strlen(str) - 1] != '1')
        valid = 0;

    if (valid)
        printf("String Accepted\n");
    else
        printf("String Rejected\n");

    return 0;
}
